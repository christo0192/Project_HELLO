# Direct runtime dependencies for the LiveKit Agents voice worker (agent.py).
#
# Pinned EXACTLY to the versions observed in the working-tree environment
# (2026-08-02, livekit-agents 1.6.4 series). This is a direct-pin file, NOT a
# full lock: transitive dependencies are resolved by pip at install time, so
# the transitive closure is not claimed to be reproducible byte-for-byte.
#
# Python compatibility (verified 2026-08-02 from public PyPI metadata,
# no login, no env values read):
#   livekit-agents            requires-python <3.15,>=3.10  -> 3.12 OK
#   livekit-plugins-sarvam    requires-python >=3.10.0      -> 3.12 OK
#   livekit-plugins-openai    requires-python >=3.10.0      -> 3.12 OK
#   python-dotenv             requires-python >=3.10        -> 3.12 OK
#   supabase                  requires-python >=3.9         -> 3.12 OK
#   httpx                     requires-python >=3.8         -> 3.12 OK
#
# Closure notes (verified from PyPI requires_dist, 2026-08-02):
#   supabase==2.31.0 pins httpx<0.29,>=0.26, so httpx==0.28.1 is in range.
#   livekit-plugins-sarvam depends on livekit-agents[codecs,openai]>=1.6.4.
livekit-agents==1.6.4
livekit-plugins-openai==1.6.4
livekit-plugins-sarvam==1.6.4
# Native Google GenAI LLM plugin (matches the 1.6.4 series). Required so the
# phone interviewer + coverage judge can speak to Gemini through the NATIVE
# google-genai SDK instead of the OpenAI-compat endpoint: Google's implicit
# prompt caching (the stable system-prompt prefix served from cache) engages
# ONLY on the native SDK, not on the OpenAI-compat shim. It pulls in
# google-genai transitively (observed google-genai==2.22.0 with this pin).
livekit-plugins-google==1.6.4
python-dotenv==1.2.2
supabase==2.31.0
httpx==0.28.1
# NOT LISTED HERE, ON PURPOSE: pyrnnoise==0.4.5 (Apache-2.0), the phone-path
# RNNoise library used by noise_suppression.py. It is installed by the
# Dockerfile builder stage as a SEPARATE step:
#   pip install --no-cache-dir --no-deps pyrnnoise==0.4.5
# A requirements line cannot carry --no-deps, and without it pip would pull
# pyrnnoise's declared deps (audiolab, matplotlib, click, tqdm), which serve
# only its CLI/wav helpers. The worker uses nothing but the librnnoise.so the
# wheel bundles (bound via ctypes; `pyrnnoise` itself is never imported), and
# the one runtime dependency that needs, numpy, already arrives unconditionally
# with livekit-agents==1.6.4 (Requires-Dist: numpy>=1.26.0). A local venv that
# skips that step simply runs with suppression fail-open OFF (logged
# `phone_noise_suppression_unavailable reason=not_installed`).
